  https://docs.optimism.io/op-stack/contribute/content-guide

  Usage:
    import { NormativeSpec } from "/snippets/normative-spec.mdx"

    <NormativeSpec
      what="The derivation pipeline"
      title="L2 chain derivation specification"
      href="https://specs.optimism.io/protocol/derivation.html"
      note="This page is a short orientation; it does not restate the spec."
    />

    what  - the singular subject the spec defines ("This upgrade").
    title - the linked spec section's name.

            to the exact section, per the cross-repo link policy.
    note  - what this page does instead of restating the spec.
*/}

export const NormativeSpec = ({ what, title, href, note }) => (
  <Info>
    <strong>규범적 명세(Normative spec):</strong> {what}은(는){" "}
    <a href={href}>{title}</a>에 규범적으로 정의되어 있습니다. {note}
  </Info>
